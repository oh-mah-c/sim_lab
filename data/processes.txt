# Sunnyworld process registry — processing recipes are DATA (TODO §9.5).
# A process declares its real conditions; the engine enforces them.
# Fails when inputs/fuel/temperature are missing — no free products.
#
# Format:
#   process <name>
#     input <material> <qty>      # consumed from targets
#     fuel <material> <qty>       # consumed from targets (optional)
#     min_temperature <deci-°C>   # best participant `temperature` must reach it
#     ticks <n>                   # simulated duration (recorded in event)
#     output <material> <qty>     # produced at actor position
#     waste <material> <qty>      # produced too — keeps mass accounted

# Smelt copper ore into metal. Needs ~1000°C (10000 deci-°C) on some
# participant (a furnace charge carries `temperature`), charcoal fuel, 24 ticks.
# Run only just at the minimum, part of the metal is lost to slag (see
# ProcessingModel efficiency); hotter charges recover more.
process smelt_copper
  input copper_ore 400
  fuel charcoal 50
  min_temperature 10000
  ticks 24
  output copper_metal 300
  waste slag 100

# Grind nitre, yellow mineral and charcoal together (no heat needed — the
# product is dangerous BECAUSE heat sets it off; see ThermalModel blast).
process grind_energetic
  input saltpeter 75
  input sulfur 10
  input charcoal 15
  ticks 6
  output energetic_mix 100


# Bronze — copper + tin alloyed in one charge. Hotter than copper alone;
# the alloy is tougher and keeps an edge (malleability 600 on `bronze`).
process smelt_bronze
  input copper_ore 200
  input tin_ore 120
  fuel charcoal 60
  min_temperature 12000
  ticks 30
  output bronze 240
  waste slag 180

# Wrought iron — the deep-mine metal. Needs a much hotter charge than
# copper/bronze and sacrifices more to slag; the payoff is tensile 900
# edge_sharpness 850 — tools and blades that out-cut stone and bone.
process smelt_iron
  input iron_ore 500
  fuel charcoal 80
  min_temperature 15500
  ticks 40
  output iron_ingot 320
  waste slag 180

# Weave fleece into cloth — no heat, just labor ticks. The product is a
# high-insulation textile: the cold-climate route from fiber_beast to
# survivable winters.
process weave_wool
  input wool 60
  ticks 20
  output wool_cloth 45
